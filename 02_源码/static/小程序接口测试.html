⑤ 小程序接入代码样例（可直接粘贴）

// 微信小程序：发起一次问诊查询
wx.request({
  url: 'http://127.0.0.1:8137/chat',   // 真机/上线请换成 https 域名（见部署说明）
  method: 'POST',
  header: { 'content-type': 'application/json' },
  data: { message: '昆明哪家医院有抗蛇毒血清？', session_id: app.globalData.sid || null },
  success: (res) => {
    const env = res.data;                 // ResponseEnvelope
    app.globalData.sid = env.session_id;  // 记住会话，实现多轮
    if (env.ok === false) {
      wx.showToast({ title: env.output.usage_tips[0] || '请求未成功', icon: 'none' });
      return;
    }
    const out = env.output;               // 4 段式
    // ① 查询条件：out.query_condition
    // ② 查询结果：out.query_results[]（name/type/description/info_status/source{title,url,source_type,updated_note}）
    // ③ 信息依据：out.info_basis[]（title/url/note/source_type）
    // ④ 使用提示：out.usage_tips[]
    // 检索留痕：env.retrieval_log[]（给"可核验"用，可不展示给用户）
    wx.setStorageSync('lastAnswer', out);
  },
  fail: (e) => wx.showToast({ title: '网络异常', icon: 'none' })
});
说明：SSE（/chat/stream）仅 Web 端调试用；小程序不便消费 SSE，统一用 /chat。 接口已开 CORS（开发环境白名单在 OPC_CORS_ORIGINS 配置）；上线微信小程序需 HTTPS + 域名白名单备案。
